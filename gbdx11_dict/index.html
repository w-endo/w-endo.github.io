▶

// 初期化
int hImage_ = -1;

void Initialize() {
    hImage_ = Image::Load("Assets/char.png");

void Draw() {
    Image::Draw(hImage_);
}

void Release() {
    Image::Release(hImage_);
}
Image::Initialize() は Main.cpp 内で自動的に呼ばれるため、GameObject のコンストラクタで呼ぶ必要はありません。

Bullet クラス（Bullet.h / Bullet.cpp）を作る

// Bullet.h
class Bullet : public GameObject {
    int hModel_;
    XMFLOAT3 dir_;    // 飛ぶ方向（正規化済み）
    int life_ = 120; // 寿命フレーム
public:
    Bullet(GameObject* p) : GameObject(p, "Bullet") {}
    void SetDirection(XMFLOAT3 d) { dir_ = d; }
    void Initialize() override;
    void Update()     override;
    void Draw()      override;
    void Release()   override;
};
02

Bullet::Update で毎フレーム前進 ＆ 寿命管理

void Bullet::Update() {
    const float SPEED = 0.5f;
    XMFLOAT3 pos = GetPosition();
    pos.x += dir_.x * SPEED;
    pos.y += dir_.y * SPEED;
    pos.z += dir_.z * SPEED;
    SetPosition(pos);

    if (--life_ <= 0) KillMe(); // 寿命切れ
}

void Bullet::OnCollision(GameObject* pTarget) override {
    if (pTarget->GetObjectName() == "Enemy") {
        KillMe(); // 敵に当たったら消える
    }
}
03

Player::Update でボタン検知 → Instantiate

if (Input::IsMouseButtonDown(0)) { // 左クリック
    Bullet* pB = Instantiate<Bullet>(this);
    pB->SetPosition(GetPosition());

    // カメラの向きを「弾の方向」として渡す
    XMFLOAT3 camPos = Camera::GetPosition();
    XMFLOAT3 camTgt = Camera::GetTarget();
    XMFLOAT3 dir = {
        camTgt.x - camPos.x,
        camTgt.y - camPos.y,
        camTgt.z - camPos.z
    };
    // 正規化
    float len = sqrtf(dir.x*dir.x + dir.y*dir.y + dir.z*dir.z);
    dir = { dir.x/len, dir.y/len, dir.z/len };
    pB->SetDirection(dir);
}

▶
📐 設計の流れ
Player と Enemy に BoxCollider→ OnCollision で Y 座標を比較→ 上から → 敵を倒す→ 横から → プレイヤーがダメージ 
// Enemy.cpp の OnCollision
void Enemy::OnCollision(GameObject* pTarget) override {
    if (pTarget->GetObjectName() != "Player") return;

    XMFLOAT3 playerPos = pTarget->GetWorldPosition();
    XMFLOAT3 myPos     = GetWorldPosition();

    // プレイヤーの足元が自分の頭より上なら「踏んだ」と判定
    if (playerPos.y > myPos.y + 0.5f) {
        // 踏まれた → 撃破エフェクト＆消滅
        EmitterData e;
        e.position = myPos;
        e.number   = 10;
        VFX::Start(e);
        KillMe();
    }
    // 横・下からの衝突はプレイヤー側の OnCollision でダメージ処理
}
プレイヤー側の OnCollision でも同様に Y 比較して、横から当たっていたらダメージを与えます。

// Enemy::Update
GameObject* pPlayer = FindObject("Player");
if (pPlayer == nullptr) return;

XMFLOAT3 myPos  = GetWorldPosition();
XMFLOAT3 target = pPlayer->GetWorldPosition();

// プレイヤーへの方向ベクトル
float dx = target.x - myPos.x;
float dz = target.z - myPos.z;
float dist = sqrtf(dx*dx + dz*dz);

if (dist > 0.1f) {
    const float SPEED = 0.05f;
    SetPosition(
        myPos.x + dx / dist * SPEED,
        myPos.y,
        myPos.z + dz / dist * SPEED
    );
    // プレイヤーの方向に向く（Y回転）
    float angle = atan2f(dx, dz);
    SetRotateY(angle);
}

void Enemy::FireRadial(int count, float speed) {
    float step = XM_2PI / count;  // 360° ÷ 弾数
    for (int i = 0; i < count; i++) {
        float angle = step * i;
        EnemyBullet* pB = Instantiate<EnemyBullet>(this);
        pB->SetPosition(GetWorldPosition());
        pB->SetVelocity(XMFLOAT3(
            cosf(angle) * speed,
            sinf(angle) * speed,
            0
        ));
    }
}

    FireRadial(8, 4.0f);
}

class Door : public GameObject {
    bool  isOpening_ = false;
    float openProgress_ = 0.0f; // 0.0 = 閉, 1.0 = 全開
public:
    void Open() { isOpening_ = true; }

    void Update() override {
        if (!isOpening_) return;
        openProgress_ = min(1.0f, openProgress_ + 0.02f);

        // 扉を上に移動させて「開く」演出
        XMFLOAT3 pos = GetPosition();
        pos.y = baseY_ + openProgress_ * 4.0f;
        SetPosition(pos);

        if (openProgress_ >= 1.0f) {
            Leave();      // 完全に開いたら Update を止める
            Invisible();  // 非表示にする（当たり判定も無効化するなら ClearCollider）
        }
    }
};

プレイヤーが箱を押して動かせるようにしたい
OnCollisionGetWorldPositionSetPosition
▶

衝突した方向を Y 座標の比較で検出し、横から押されたときだけ箱を動かします。

// Box::OnCollision
void Box::OnCollision(GameObject* pTarget) override {
    if (pTarget->GetObjectName() != "Player") return;

    XMFLOAT3 myPos     = GetWorldPosition();
    XMFLOAT3 playerPos = pTarget->GetWorldPosition();

    // 上から乗っている場合は押さない（Y差が大きい）
    if (playerPos.y > myPos.y + 0.8f) return;

    // プレイヤーから箱への水平方向に押し出す
    float dx = myPos.x - playerPos.x;
    float dz = myPos.z - playerPos.z;
    float len = sqrtf(dx*dx + dz*dz);
    if (len < 0.001f) return;

    const float PUSH = 0.1f;
    SetPosition(
        myPos.x + dx/len * PUSH,
        myPos.y,
        myPos.z + dz/len * PUSH
    );
}

void GameScene::Initialize() {
    CsvReader csv;
    csv.Load("Assets/stage01.csv");

    const float CELL = 2.0f; // 1セルのワールドサイズ

    for (int y = 0; y < (int)csv.GetHeight(); y++) {
        for (int x = 0; x < (int)csv.GetWidth(); x++) {
            int val = csv.GetValue(x, y);
            XMFLOAT3 pos = { x*CELL, 0, y*CELL };
            switch (val) {
            case 1: {
                Floor* p = Instantiate<Floor>(this);
                p->SetPosition(pos); break;
            }
            case 2: {
                Switch* p = Instantiate<Switch>(this);
                p->SetPosition(pos); break;
            }
            case 3: {
                Goal* p = Instantiate<Goal>(this);
                p->SetPosition(pos); break;
            }
            }
        }
    }
